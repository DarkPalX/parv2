	Copy PARv2 dept in accountabilityHeaders to PARv1 dept in employee
	QUERY:
	-- FOR VIEWING FIRST ⚠️
		SELECT DISTINCT
			e.dept COLLATE DATABASE_DEFAULT AS employee_dept,
			CASE
				WHEN e.dept IS NULL THEN NULL
				WHEN e.dept NOT LIKE '%[A-Z]%' THEN e.dept COLLATE DATABASE_DEFAULT
				ELSE a.dept COLLATE DATABASE_DEFAULT
			END AS suggested_dept,
			a.similarity_score
		FROM parv1_employee e
		OUTER APPLY (
			SELECT TOP 1
				ah.dept,
				DIFFERENCE(
					UPPER(LTRIM(RTRIM(e.dept))) COLLATE DATABASE_DEFAULT,
					UPPER(LTRIM(RTRIM(ah.dept))) COLLATE DATABASE_DEFAULT
				) AS similarity_score
			FROM accountabilityHeaders ah
			WHERE ah.dept IS NOT NULL
				AND LTRIM(RTRIM(ah.dept)) <> ''
			ORDER BY similarity_score DESC
		) a
		ORDER BY e.dept COLLATE DATABASE_DEFAULT


	--FOR UPDATING ⚠️
		UPDATE e
		SET dept = a.dept COLLATE DATABASE_DEFAULT
		FROM parv1_employee e
		OUTER APPLY (
			SELECT TOP 1
				ah.dept,
				DIFFERENCE(
					UPPER(LTRIM(RTRIM(e.dept))) COLLATE DATABASE_DEFAULT,
					UPPER(LTRIM(RTRIM(ah.dept))) COLLATE DATABASE_DEFAULT
				) AS similarity_score
			FROM accountabilityHeaders ah
			WHERE ah.dept IS NOT NULL
			AND LTRIM(RTRIM(ah.dept)) <> ''
			ORDER BY similarity_score DESC
		) a
		WHERE e.dept IS NOT NULL
		AND e.dept LIKE '%[A-Z]%'   -- skip numbers / garbage
		AND a.similarity_score >= 3; -- optional safety threshold
